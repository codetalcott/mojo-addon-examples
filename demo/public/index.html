One Million Numbers, One Call

Compute mean, stddev, min, max, and percentiles across 1M random Float64 values — with a 50-bin histogram — in a single Mojo call.

Computing statistics...
fn _stats_kernel(
    data: UnsafePointer[Float64, MutAnyOrigin],
    n: Int,
) -> StatResult:
    var sum: Float64 = 0.0
    var min_val = data[0]
    var max_val = data[0]

    # SIMD reduction: process 4 Float64s per cycle
    fn reduce[width: Int](i: Int) unified {mut}:
        var chunk = data.load[width=width](i)
        sum += chunk.reduce_add()
        min_val = min(min_val, chunk.reduce_min())
        max_val = max(max_val, chunk.reduce_max())
    vectorize[simd_width_of[DType.float64]()](n, reduce)

fn _count_byte_simd(
    data: UnsafePointer[Byte, MutAnyOrigin],
    n: Int, target: Byte,
) -> Int:
    var count: Int = 0
    # Compare 64 bytes at once using SIMD
    fn scan[width: Int](i: Int) unified {mut}:
        var chunk = data.load[width=width](i)
        var matches = chunk == SIMD[DType.uint8, width](target)
        count += Int(matches.cast[DType.uint8]().reduce_add())
    vectorize[64](n, scan)
    return count

# SIMD vectorized inner loop — process multiple columns at once
fn _matmul_vectorized(a, b, c, M, K, N):
    for i in range(M):
        var row_c = i * N
        for p in range(K):
            var a_ip = a[i * K + p]
            var row_b = p * N
            fn compute[width: Int](j: Int) unified {mut}:
                var b_chunk = b.load[width=width](row_b + j)
                var c_chunk = c.load[width=width](row_c + j)
                c.store[width=width](row_c + j, c_chunk + a_ip * b_chunk)
            vectorize[simd_width_of[DType.float64]()](N, compute)

# 48-byte main loop — process 3 x 16-byte blocks per iteration
fn _wyhash_core(data: UnsafePointer[Byte, MutAnyOrigin],
                len: Int, seed: UInt64) -> UInt64:
    var see1 = seed
    var see2 = seed
    var p = 0
    while len - p > 48:
        seed = wymum(wyr8(data, p) ^ WYP1, wyr8(data, p + 8) ^ seed)
        see1 = wymum(wyr8(data, p + 16) ^ WYP2, wyr8(data, p + 24) ^ see1)
        see2 = wymum(wyr8(data, p + 32) ^ WYP3, wyr8(data, p + 40) ^ see2)
        p += 48
    return wymum(WYP1 ^ UInt64(len), wymum(a ^ WYP1, b ^ seed))
